ESP32 Arduino Firmware (BLE & Buzzer Code)

This code runs on ESP32 development board. Connect a passive/active buzzer to GPIO pin 18/23. Flash with Arduino IDE.

/* ChronoSync ESP32 BLE Smart Alarm Firmware */
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

#define BUZZER_PIN 18 // GPIO connected to buzzer (+)
#define SERVICE_UUID "6E400001-B5A3-F393-E0A9-E50E24DCCA9E"
#define CHAR_RX_UUID "6E400002-B5A3-F393-E0A9-E50E24DCCA9E"
#define CHAR_TX_UUID "6E400003-B5A3-F393-E0A9-E50E24DCCA9E"

bool alarmActive = false;
String currentPattern = "rapid";

void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);
  
  BLEDevice::init("ChronoSync_Alarm_BLE");
  BLEServer *pServer = BLEDevice::createServer();
  BLEService *pService = pServer->createService(SERVICE_UUID);
  // Full sketch provided in repository: arduino_esp32_smart_alarm.ino
}

void loop() {
  if (alarmActive) {
    // Jaldi-Jaldi On/Off Rapid Buzzer:
    digitalWrite(BUZZER_PIN, HIGH);
    delay(100);
    digitalWrite(BUZZER_PIN, LOW);
    delay(100);
  }
}
